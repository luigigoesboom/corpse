extends Creature
## {{title}}: (what it is, where it lives, how to beat it). Made with
## tools/creature/new_creature.gd as a flyer; its look is
## tools/pixel/recipes/creature_{{kind}}.gd.
##
##   HOVER    drift around its home, look for the player
##   STALK    fly to a point above and beside the player
##   WINDUP   stop and shake (the tell)
##   DIVE     dash at where the player was, the Attack hitbox on
##   RECOVER  climb back up

enum State { HOVER, STALK, WINDUP, DIVE, RECOVER }

@export var speed := 130.0
@export var dive_speed := 400.0
@export var windup_time := 0.45

var _home := Vector2.ZERO
var _dive := Vector2.ZERO


func setup() -> void:
	_home = position


func think(delta: float) -> void:
	var want := Vector2.ZERO
	match state:
		State.HOVER:
			want = (_home + Vector2(sin(state_time * 1.3) * 40.0, sin(state_time * 2.1) * 16.0) - position) * 2.0
			if sees_player():
				enter(State.STALK)
		State.STALK:
			if not player_alive() or to_player().length() > lose_range:
				enter(State.HOVER)
				return
			var side := -signf(to_player().x) if to_player().x != 0.0 else 1.0
			var perch := player.global_position + Vector2(side * 90.0, -100.0)
			want = (perch - global_position).limit_length(speed)
			face(to_player().x)
			if global_position.distance_to(perch) < 30.0 or state_time > 3.0:
				enter(State.WINDUP)
		State.WINDUP:
			visual.position.x = sin(state_time * 80.0)
			if state_time > windup_time:
				visual.position.x = 0.0
				_dive = to_player().normalized() if player_alive() else Vector2.DOWN
				strike(0.6)
				enter(State.DIVE)
		State.DIVE:
			velocity = _dive * dive_speed
			if state_time > 0.6 or get_slide_collision_count() > 0:
				enter(State.RECOVER)
			return
		State.RECOVER:
			want = Vector2(0, -speed)
			if state_time > 0.7:
				enter(State.STALK)
	velocity = velocity.lerp(want, minf(1.0, delta * 8.0))


func animation() -> StringName:
	if is_dead():
		return &"die"
	if hurt_time > 0.0:
		return &"hurt"
	match state:
		State.WINDUP:
			return &"windup"
		State.DIVE:
			return &"attack"
	return &"move"


func hurt(_amount: int, from: Vector2, knockback: float) -> void:
	velocity = (global_position - from).normalized() * knockback
	if state in [State.HOVER, State.WINDUP]:
		enter(State.RECOVER)
