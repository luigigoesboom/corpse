extends Creature
## {{title}}: (what it is, where it lives, how to beat it). Made with
## tools/creature/new_creature.gd as a rooted thing; its look is
## tools/pixel/recipes/creature_{{kind}}.gd.
##
##   IDLE     sway, wait for the player to come into reach
##   WINDUP   turn and draw back (the tell)
##   ATTACK   strike with the Attack hitbox on
##   REST     a pause to punish it

enum State { IDLE, WINDUP, ATTACK, REST }

@export var reach := 70.0
@export var windup_time := 0.5
@export var rest_time := 1.0


func think(delta: float) -> void:
	velocity.x = move_toward(velocity.x, 0.0, 800.0 * delta)
	match state:
		State.IDLE:
			if sees_player(reach, 64.0):
				face(to_player().x)
				enter(State.WINDUP)
		State.WINDUP:
			if state_time > windup_time:
				strike(0.2)
				enter(State.ATTACK)
		State.ATTACK:
			if state_time > 0.3:
				enter(State.REST)
		State.REST:
			if state_time > rest_time:
				enter(State.IDLE)


func animation() -> StringName:
	if is_dead():
		return &"die"
	if hurt_time > 0.0:
		return &"hurt"
	match state:
		State.WINDUP:
			return &"windup"
		State.ATTACK:
			return &"attack"
	return &"idle"
